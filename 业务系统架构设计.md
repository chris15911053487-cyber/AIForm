### 1.3 技术栈选型说明

| 层级 | 技术选型 | 选型理由 |
|------|----------|----------|
| **Web前端** | Vue3 + Element Plus | 生态成熟、组件丰富、学习成本低 |
| **移动端** | uni-app (Vue3) | 跨平台能力强、一套代码多端运行 |
| **后端框架** | NestJS | 模块化架构、装饰器语法、TypeScript原生支持 |
| **主数据库** | PostgreSQL | 功能丰富、JSON支持好、性能稳定 |
| **缓存** | Redis | 性能卓越、支持多种数据结构 |
| **消息队列** | BullMQ | 基于Redis、Job队列、延迟任务支持 |
| **AI编排** | LangChain | 主流LLM集成、Chain机制、RAG支持 |
| **向量数据库** | Qdrant/Pgvector | 轻量部署、与PG兼容性好 |
| **文件存储** | MinIO | S3兼容、私有部署成本低 |
| **API网关** | Kong/Nginx | 成熟稳定、插件丰富 |
| **容器化** | Docker + Docker Compose | 环境一致、快速部署 |
| **CI/CD** | GitHub Actions | GitHub原生、免费额度 |

### 2.9 各服务降级方案

| 服务 | 不可用时表现 | 用户提示 | 恢复后处理 |
|-----|------------|---------|----------|
| ai-service | AI问答功能返回错误 | "AI服务暂不可用，请稍后再试" | 自动恢复，无需操作 |
| knowledge-service | 知识库搜索返回空 | "知识库服务维护中" | 自动恢复 |
| skill-service | Skill调用失败 | "技能执行服务暂不可用" | 自动恢复 |

| workflow-service | 审批功能暂停 | "审批服务维护中，请稍后提交" | 恢复后继续处理 |

### 2.10 后端降级实现

#### 服务调用降级示例

```typescript
// 使用Circuit Breaker模式实现服务降级
import { CircuitBreaker } from 'opossum';

const breakerOptions = {
  timeout: 3000,      // 超时时间
  errorThresholdPercentage: 50,  // 错误率阈值
  resetTimeout: 10000,  // 重置时间
};

const aiServiceBreaker = new CircuitBreaker(aiServiceCall, breakerOptions);

// 降级处理
aiServiceBreaker.fallback((error) => {
  return {
    success: false,
    degraded: true,
    message: 'AI服务暂不可用，请稍后再试',
    data: null
  };
});

// 熔断事件记录
aiServiceBreaker.on('open', () => console.log('AI服务熔断器打开'));
aiServiceBreaker.on('close', () => console.log('AI服务熔断器关闭'));
```

#### API网关降级配置

```yaml
# Nginx降级配置示例
upstream core_backend {
    server core-api:3000;
}

upstream ai_backend {
    server ai-service:3002;
    server ai-service-backup:3002 backup;
}

server {
    location /api/ai/ {
        proxy_pass http://ai_backend;
        
        # 降级配置
        proxy_intercept_errors on;
        error_page 502 503 504 = @ai_fallback;
    }
    
    location @ai_fallback {
        default_type application/json;
        return 200 '{"success":false,"message":"AI服务暂不可用，请稍后再试"}';
    }
}
```

### 2.11 前端降级处理

```typescript
// 前端API调用封装
class ApiClient {
  async callService(service: string, endpoint: string, options?: RequestInit) {
    try {
      const response = await fetch(`${service}${endpoint}`, {
        ...options,
        signal: AbortSignal.timeout(10000)
      });
      
      if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
      }
      
      return await response.json();
    } catch (error) {
      // 服务不可用时返回降级数据
      return this.handleDegradation(service, error);
    }
  }
  
  private handleDegradation(service: string, error: Error) {
    const degradationMap = {
      'ai-service': {
        success: false,
        message: 'AI服务暂不可用，请稍后再试',
        degraded: true
      },
      'knowledge-service': {
        success: true,
        data: [],
        message: '知识库服务维护中',
        degraded: true
      },
      'report-service': {
        success: false,
        message: '报表服务暂不可用',
        degraded: true
      },
      'message-service': {
        success: true,
        polling: true,
        message: '消息服务维护中，已切换轮询模式',
        degraded: true
      },
      'workflow-service': {
        success: false,
        message: '审批服务维护中，请稍后提交',
        degraded: true
      },
      'skill-service': {
        success: false,
        message: '技能执行服务暂不可用',
        degraded: true
      }
    };
    
    return degradationMap[service] || {
      success: false,
      message: '服务暂不可用'
    };
  }
}
```

#### 前端动态路由方案

```
┌─────────────────────────────────────────────────────────────────────────┐
│                          前端动态路由加载流程                              │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                         │
│  1. 用户登录成功                                                         │
│         ↓                                                               │
│  2. 调用 GET /api/menus/routes 获取菜单树                                │
│         ↓                                                               │
│  3. 前端解析菜单数据，生成Vue Router配置                                  │
│         ↓                                                               │
│  4. 使用 router.addRoute() 动态添加路由                                  │
│         ↓                                                               │
│  5. 支持懒加载组件（() => import('path'))                                │
│                                                                         │
└─────────────────────────────────────────────────────────────────────────┘
```

**前端路由生成示例：**

```typescript
// 后端返回的菜单数据
const menuData = [
  {
    id: '1',
    name: 'Dashboard',
    path: '/dashboard',
    component: '/dashboard/index',
    meta: { title: '首页', icon: 'home' }
  },
  {
    id: '2',
    name: 'UserManage',
    path: '/system',
    children: [
      {
        id: '2-1',
        name: 'UserList',
        path: '/system/users',
        component: '/system/users/index',
        meta: { title: '用户管理' }
      }
    ]
  }
];

// 转换为Vue Router配置
function generateRoutes(menus: MenuRoute[]): RouteRecordRaw[] {
  return menus.map(menu => ({
    path: menu.path,
    name: menu.name,
    component: menu.component 
      ? () => import(`@/views${menu.component}.vue`)  // 懒加载
      : undefined,
    meta: menu.meta,
    redirect: menu.redirect,
    children: menu.children ? generateRoutes(menu.children) : undefined
  }));
}
```

#### 与权限模块的关联

```
┌─────────────────────────────────────────────────────────────────┐
│                      权限体系关联                                 │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│   角色 (Role)  ──── 角色权限 ──── 资源 (Resource)                │
│       │                              │                          │
│       │                              │ type='menu'              │
│       │                              ↓                          │
│       │                      ┌──────────────┐                   │
│       │                      │   菜单实体   │                   │
│       │                      │   (Menu)    │                   │
│       │                      └──────────────┘                   │
│       │                              │                          │
│       └──────────────────────────────┘                          │
│                        权限校验时双重检查                          │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

- 菜单作为一种权限资源，在权限模块的`resource`表中`type='menu'`
- 角色权限分配时，可以选择菜单类型的资源
- 前端获取菜单时，只返回当前用户有权限的菜单
- 菜单权限校验通过后，检查用户角色是否包含该菜单

```typescript
    this.skills.set(skill.id, skill);

  private wrapAsTool(skill: Skill): Tool {

        return JSON.stringify(result);
      }
    });
  }
  
  // 获取所有Tools供Agent使用
  getTools(): Tool[] {
    return Array.from(this.skillTools.values());
  }
```

```sql
-- 日志归档策略
CREATE OR REPLACE FUNCTION archive_old_logs()
RETURNS void AS $$
BEGIN
    -- 6个月前的操作日志归档到历史表
    INSERT INTO log_operations_archive 
    SELECT * FROM log_operations 
    WHERE timestamp < CURRENT_TIMESTAMP - INTERVAL '6 months';
    
    DELETE FROM log_operations 
    WHERE timestamp < CURRENT_TIMESTAMP - INTERVAL '6 months';
    
    -- 1年前的数据转储到冷存储
    DELETE FROM log_operations_archive 
    WHERE timestamp < CURRENT_TIMESTAMP - INTERVAL '1 year';
END;
$$ LANGUAGE plpgsql;

-- 创建归档表
CREATE TABLE log_operations_archive (
    LIKE log_operations INCLUDING ALL
);

-- 设置保留策略
ALTER TABLE log_operations SET (
    timescaledb.archive_chunks = 12,
    timescaledb.ignore_invalidation_older_than = '1 day'
);
```

```yaml
      - ENCRYPTION_KEY=${ENCRYPTION_KEY}

      - db

      - "5432:5432"
    networks:

    restart: unless-stopped

  redis:
    image: redis:7-alpine

    volumes:

      - "6379:6379"

      - "6334:6334"
```

## 7. 分批上线部署规划

为降低上线风险、节省初期成本，建议采用分批上线策略，将系统分为三个阶段逐步部署。

### 7.1 第一阶段：核心功能（MVP最小可用版本）

**部署容器：**

| 容器 | 功能 |
|------|------|
| api-gateway | API网关 |
| core-api | 用户权限、菜单管理、表单引擎、数据源管理 |
| PostgreSQL | 数据库 |
| Redis | 缓存 |

**服务器配置：**
- 单台服务器：4核8G，100GB SSD
- 或与现有服务器共存

**上线后能力：**
- 用户登录、注册、权限管理
- 菜单配置、动态路由
- 表单创建、填报、查询
- 基础数据操作

**验证目标：**
- 用户能正常登录和使用基础功能
- 表单流程顺畅
- 系统稳定性验证

---

### 7.2 第二阶段：业务增强

**新增部署容器：**

| 容器 | 功能 |
|------|------|
| report-service | 报表引擎、报表配置 |
| message-service | 消息中心、WebSocket推送、邮件融合 |
| workflow-service | 审批流程、流程设计器 |

**服务器配置：**
- 原服务器扩容至8核16G
- 或新增一台4核8G业务服务器

**上线后能力：**
- 自定义报表、导出、分析
- 系统消息、审批通知
- 多级审批流程
- 邮件与消息融合

**验证目标：**
- 报表生成和导出正常
- 消息推送及时准确
- 审批流程跑通

---

### 7.3 第三阶段：AI能力

**新增部署容器：**

| 容器 | 功能 |
|------|------|
| ai-service | AI问答、多模型接入、截图识别 |
| knowledge-service | 知识库管理、向量化检索 |
| skill-service | Skill上传、执行沙箱 |
| 向量数据库 | Milvus或Qdrant |

**服务器配置：**
- 独立AI服务器：8核16G，100GB SSD
- 向量库需要足够内存和存储

**上线后能力：**
- 智能问答（公司制度、操作指南）
- 知识库检索
- Skill自动/手动调用
- 截图识别和处理

**验证目标：**
- AI回答准确率达标
- 知识库检索响应快
- Skill执行稳定

---

```
┌─────────┐         ┌─────────┐         ┌─────────┐

│         │         │         │         │         │
```

---

### 7.5 分批上线优势

| 优势 | 说明 |
|-----|-----|
| 降低风险 | 先验证核心功能，出问题影响范围小 |
| 节省成本 | AI服务最吃资源，可后期按需部署 |
| 按需选择 | 客户不需要AI功能就不部署第三阶段 |
| 灵活扩展 | 业务量增大后可独立扩展某个服务 |
| 渐进验证 | 每阶段验证通过后再进入下一阶段 |

---

### 7.6 回滚方案

每个阶段部署后出现问题：
1. 停止新部署的容器
2. 核心服务继续运行
3. 排查问题后重新部署

Docker Compose支持快速回滚到上一版本镜像。

## 8. CI/CD流程

### 8.1 GitHub Actions 工作流配置

```yaml
# .github/workflows/ci-cd.yml

name: CI/CD Pipeline

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

env:
  NODE_VERSION: '18'
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  # ===================
  # 代码检查
  # ===================
  lint:
    name: Lint & Type Check
    runs-on: ubuntu-latest
    
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
          cache-dependency-path: backend/package-lock.json
      
      - name: Install dependencies
        run: npm ci
      
      - name: Run ESLint
        run: npm run lint
      
      - name: Run Type Check
        run: npm run type-check

  # ===================
  # 单元测试
  # ===================
  test:
    name: Unit Tests
    runs-on: ubuntu-latest
    
    services:
      postgres:
        image: postgres:15
        env:
          POSTGRES_PASSWORD: postgres
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 5432:5432
      
      redis:
        image: redis:7
        options: >-
          --health-cmd "redis-cli ping"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 6379:6379
    
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
          cache-dependency-path: backend/package-lock.json
      
      - name: Install dependencies
        run: npm ci
      
      - name: Run tests
        env:
          DATABASE_URL: postgresql://postgres:postgres@localhost:5432/test
          REDIS_URL: redis://localhost:6379
        run: npm run test:cov

  # ===================
  # 构建镜像
  # ===================
  build:
    name: Build Docker Image
    runs-on: ubuntu-latest
    needs: [lint, test]
    
    outputs:
      image-tag: ${{ steps.meta.outputs.tags }}
    
    steps:
      - uses: actions/checkout@v4
      
      - name: Extract metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}
          tags: |
            type=ref,event=branch
            type=sha,prefix={{branch}}-
            type=raw,value=latest,enable={{is_default_branch}}
      
      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3
      
      - name: Login to Container Registry
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}
      
      - name: Build and push
        uses: docker/build-push-action@v5
        with:
          context: ./backend
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          cache-from: type=gha
          cache-to: type=gha,mode=max

  # ===================
  # 部署到开发环境
  # ===================
  deploy-dev:
    name: Deploy to Development
    runs-on: ubuntu-latest
    needs: build
    if: github.ref == 'refs/heads/develop'
    environment: development
    
    steps:
      - name: Deploy to server
        uses: appleboy/ssh-action@master
        with:
          host: ${{ secrets.DEV_HOST }}
          username: ${{ secrets.DEV_USER }}
          key: ${{ secrets.DEV_SSH_KEY }}
          script: |
            cd /app
            docker-compose pull
            docker-compose up -d
            docker image prune -f

  # ===================
  # 部署到生产环境
  # ===================
  deploy-prod:
    name: Deploy to Production
    runs-on: ubuntu-latest
    needs: build
    if: github.ref == 'refs/heads/main'
    environment: production
    
    steps:
      - name: Deploy to server
        uses: appleboy/ssh-action@master
        with:
          host: ${{ secrets.PROD_HOST }}
          username: ${{ secrets.PROD_USER }}
          key: ${{ secrets.PROD_SSH_KEY }}
          script: |
            cd /app
            docker-compose pull
            docker-compose up -d
            # 健康检查
            sleep 30
            curl -f https://api.yourdomain.com/health || exit 1
            docker image prune -f

  # ===================
  # 安全扫描
  # ===================
  security:
    name: Security Scan
    runs-on: ubuntu-latest
    needs: build
    
    steps:
      - uses: actions/checkout@v4
      
      - name: Run Trivy vulnerability scanner
        uses: aquasecurity/trivy-action@master
        with:
          image-ref: '${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:${{ github.sha }}'
          format: 'sarif'
          output: 'trivy-results.sarif'
      
      - name: Upload Trivy scan results
        uses: github/codeql-action/upload-sarif@v2
        with:
          sarif_file: 'trivy-results.sarif'
```

### 8.3 零停机部署

生产环境使用 Docker 滚动更新实现零停机：

```yaml
# docker-compose 生产部署配置
services:
  core-api:
    deploy:
      replicas: 2
      update_config:
        parallelism: 1
        delay: 10s
        order: start-first
        failure_action: rollback
```

部署流程：
1. 启动新版本容器并等待健康检查通过
2. 旧容器停止接收新请求（通过 Nginx upstream 摘除）
3. 旧容器处理完现有请求后关闭
4. Nginx 自动将流量切换到新容器

要求：服务必须支持多实例无状态运行（所有状态由 Redis/PostgreSQL 管理）。

```typescript
  const prefix = 'sk';
  const randomPart = crypto.randomBytes(32).toString('hex');
  return `${prefix}_${randomPart}`;

async function storeAPIKey(key: string, config: APIKeyConfig): Promise<void> {
  const hash = crypto.createHash('sha256').update(key).digest('hex');
  config.keyHash = hash;
  await db.apiKeys.save(config);
}

async function validateAPIKey(key: string): Promise<boolean> {
  const hash = crypto.createHash('sha256').update(key).digest('hex');
  const config = await db.apiKeys.findByHash(hash);
  return config && config.status === 'active' && (!config.expiresAt || config.expiresAt > new Date());
}
```

### 9.3 敏感数据加密

```typescript
// 加密策略
const encryptionConfig = {
  algorithm: 'aes-256-gcm',
  keyDerivation: 'pbkdf2',
  iterations: 100000,
  keyLength: 32,
  ivLength: 16,
  saltLength: 32,
  tagLength: 16,
};

// 需要加密的字段
const sensitiveFields = [
  'password',
  'apiKey',
  'secretKey',
  'accessToken',
  'refreshToken',
  'privateKey',
  'certificate',
];

// 加密服务
class EncryptionService {
  private key: Buffer;
  
  constructor() {
    this.key = crypto.scryptSync(
      process.env.ENCRYPTION_KEY,
      'salt',
      32
    );
  }
  
  encrypt(plaintext: string): EncryptedData {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv('aes-256-gcm', this.key, iv);
    
    let encrypted = cipher.update(plaintext, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    
    return {
      encrypted,
      iv: iv.toString('hex'),
      tag: cipher.getAuthTag().toString('hex'),
    };
  }
  
  decrypt(data: EncryptedData): string {
    const decipher = crypto.createDecipheriv(
      'aes-256-gcm',
      this.key,
      Buffer.from(data.iv, 'hex')
    );
    decipher.setAuthTag(Buffer.from(data.tag, 'hex'));
    
    let decrypted = decipher.update(data.encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    
    return decrypted;
  }
}

// 字段级加密装饰器
function Encrypted(target: any, propertyKey: string) {
  let value: string;
  
  Object.defineProperty(target, propertyKey, {
    get: () => value,
    set: (newValue) => {
      if (newValue && typeof newValue === 'string') {
        value = encryptionService.encrypt(newValue).encrypted;
      } else {
        value = newValue;
      }
    },
    enumerable: true,
    configurable: true,
  });
}
```

### 9.4 操作日志记录

```typescript
// 日志记录拦截器
@Injectable()
export class AuditInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const { method, path, body, user, ip } = request;
    
    // 记录请求
    const startTime = Date.now();
    
    return next.handle().pipe(
      tap({
        next: () => {
          // 成功时记录
          this.logOperation({
            method,
            path,
            userId: user?.id,
            username: user?.username,
            status: 'success',
            duration: Date.now() - startTime,
            ip,
          });
        },
        error: (error) => {
          // 失败时记录
          this.logOperation({
            method,
            path,
            userId: user?.id,
            username: user?.username,
            status: 'error',
            error: error.message,
            duration: Date.now() - startTime,
            ip,
          });
        },
      })
    );
  }
}

// 敏感数据脱敏
function sanitizeData(data: any, sensitiveFields: string[]): any {
  if (!data) return data;
  
  const sanitized = Array.isArray(data) ? [] : {};
  
  for (const [key, value] of Object.entries(data)) {
    if (sensitiveFields.includes(key)) {
      sanitized[key] = '***MASKED***';
    } else if (typeof value === 'object' && value !== null) {
      sanitized[key] = sanitizeData(value, sensitiveFields);
    } else {
      sanitized[key] = value;
    }
  }
  
  return sanitized;
}
```
